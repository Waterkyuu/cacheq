Ordinary timings: median of five samples; 300ms per sample; profiles and traces collected separately
Investigation was performed in a temporary git-archive checkout. Production code was not changed.

Findings:
- Shared Client state is a bottleneck: SharedHit 34.271 us vs IsolatedHit 21.585 us.
- Persistent workers barely improve these batches: SharedHit 33.429 us, FirstLoad 65.532 us.
- NoOp runs the same batch harness and digest checks without a Fetch: 2.719 us.
- FixedClockHit removes both time.Now calls under the lock: 30.290 us.
- Passing a nil fetcher only after warm-up removes its erased wrapper: 32.955 us,
  with 107 allocations instead of 207; this is a warm-cache-only diagnostic control.
- Direct freshness calculation removes the full snapshot, duplicate age/map checks,
  pending lookup, and second clock read: SharedHit 25.639 us, FirstLoad 58.078 us.
- Restoring the original query.go returns SharedHit to 35.034 us and FirstLoad to 67.871 us.
- The direct-freshness candidate passed go test -race ./... -count=1 in the temporary checkout.
- Trace: CacheHit has 20.43 Client-mutex blocking events per batch.
- Trace: FirstLoad has about 4 pending waits per batch; median time from the load waking
  a caller to that caller resuming is 5.824 us. Its 18.91 Client-mutex blocking events
  per batch mostly occur after the caller has been woken by Client.execute
  (18,623 of 18,927 events). Completed pending reads return directly without reacquiring
  the Client lock; these later waits belong to subsequent cached reads.

Interpretation limits:
- Independent Clients isolate sharing of Client state; they do not implement shared caching.
- Persistent workers exchange goroutine creation for a dispatch channel. This is a useful
  control, not exact subtraction of goroutine creation time.
- NoOp measures a lighter workload. Its cost cannot simply be subtracted from other rows.
- Shorter critical sections reduce lock contention nonlinearly. Do not sum the timing
  differences or assign percentages of the original serial/concurrent gap to each control.
- A mutex profile attributes accumulated contention to the releasing Unlock stack;
  it does not say Unlock itself is expensive, or describe a percentage of batch elapsed time.
- Trace and profile instrumentation changes timing. Aggregate waits across callers overlap
  and must not be added to infer batch latency. The original benchmark tables are unchanged.

import (

// readDiagnosis selects one controlled change to the same-key read workload.
type readDiagnosis struct {
	// mode selects cached reads, one cold load, or the harness without any cache access.
	mode string
	// persistent moves request goroutine creation outside the measured batches.
	persistent bool
	// isolated gives each caller its own warm Client to remove shared Client lock contention.
	isolated bool
	// fixedClock removes time.Now calls while preserving a fresh cached result.
	fixedClock bool
}

// BenchmarkReadDiagnosis separates batch setup, shared Client contention, and clock work.
// These controlled workloads diagnose costs and do not replace the public benefit benchmarks.
func BenchmarkReadDiagnosis(b *testing.B) {
	for _, harness := range []string{"FreshWorkers", "PersistentWorkers"} {
		b.Run(harness, func(b *testing.B) {
			for _, scenario := range []string{"NoOp", "SharedHit", "IsolatedHit", "FixedClockHit", "NilFetcherHit", "FirstLoad"} {
				b.Run(scenario, func(b *testing.B) {
					mode := "CacheHit"
					if scenario == "NoOp" || scenario == "FirstLoad" || scenario == "NilFetcherHit" {
						mode = scenario
					}
					benchmarkReadDiagnosis(b, readDiagnosis{
						mode: mode, persistent: harness == "PersistentWorkers",
						isolated: scenario == "IsolatedHit", fixedClock: scenario == "FixedClockHit",
					})
				})
			}
		})
	}
}

// benchmarkReadDiagnosis keeps 100 checked reads and four callers in every control.
func benchmarkReadDiagnosis(b *testing.B, config readDiagnosis) {
	payload := make([]byte, benchmarkPayloadBytes)
	for i := range payload {
		payload[i] = byte(i % 251)
	}
	want := sha256.Sum256(payload)
	ctx := context.Background()
	var calls atomic.Uint64
	load := func(ctx context.Context) ([sha256.Size]byte, error) {
		if err := ctx.Err(); err != nil {
			return [sha256.Size]byte{}, err
		}
		calls.Add(1)
		return sha256.Sum256(payload), nil
	}
	options := cacheq.Options{StaleTime: time.Hour}
	if config.fixedClock {
		now := time.Unix(1, 0)
		options.Clock = func() time.Time { return now }
	}
	clients := make([]*cacheq.Client, benchmarkConcurrentWorkers)
	clients[0] = cacheq.NewClient(options)
	b.Cleanup(clients[0].Close)
	for worker := 1; worker < benchmarkConcurrentWorkers; worker++ {
		clients[worker] = clients[0]
		if config.isolated {
			clients[worker] = cacheq.NewClient(options)
			b.Cleanup(clients[worker].Close)
		}
	}
	if config.mode == "CacheHit" || config.mode == "NilFetcherHit" {
		for _, client := range clients {
			if value, err := cacheq.Fetch(ctx, client, "product:1", load); err != nil || value != want {
				b.Fatalf("warm-up failed: %v", err)
			}
		}
		calls.Store(0)
	}
	var ready, finished sync.WaitGroup
	results := make([][sha256.Size]byte, benchmarkConcurrentRequests)
	failures := make([]error, benchmarkConcurrentRequests)
	run := func(worker int, start <-chan struct{}) {
		ready.Done()
		<-start
		for request := worker; request < benchmarkConcurrentRequests; request += benchmarkConcurrentWorkers {
			if config.mode == "NoOp" {
				results[request] = want
			} else if config.mode == "NilFetcherHit" {
				// Warm-cache reads do not need a loader; this control avoids allocating its erased wrapper.
				results[request], failures[request] = cacheq.Fetch[[sha256.Size]byte](ctx, clients[worker], "product:1", nil)
			} else {
				results[request], failures[request] = cacheq.Fetch(ctx, clients[worker], "product:1", load)
			}
		}
		finished.Done()
	}
	jobs := make([]chan (<-chan struct{}), benchmarkConcurrentWorkers)
	var stopped sync.WaitGroup
	if config.persistent {
		stopped.Add(benchmarkConcurrentWorkers)
		for worker := range benchmarkConcurrentWorkers {
			jobs[worker] = make(chan (<-chan struct{}), 1)
			go func() {
				defer stopped.Done()
				for start := range jobs[worker] {
					run(worker, start)
				}
			}()
		}
		b.Cleanup(func() {
			for _, job := range jobs {
				close(job)
			}
			stopped.Wait()
		})
	}
	b.ReportAllocs()
	b.ResetTimer()
	for range b.N {
		if config.mode == "FirstLoad" {
			b.StopTimer()
			clients[0].Clear()
			b.StartTimer()
		}
		results = make([][sha256.Size]byte, benchmarkConcurrentRequests)
		failures = make([]error, benchmarkConcurrentRequests)
		start := make(chan struct{})
		ready.Add(benchmarkConcurrentWorkers)
		finished.Add(benchmarkConcurrentWorkers)
		for worker := range benchmarkConcurrentWorkers {
			if config.persistent {
				jobs[worker] <- start
			} else {
				go run(worker, start)
			}
		}
		ready.Wait()
		close(start)
		finished.Wait()
		for request, err := range failures {
			if err != nil || results[request] != want {
				b.Fatalf("request %d failed: %v", request, err)
			}
		}
	}
	b.StopTimer()
	var expected uint64
	if config.mode == "FirstLoad" {
		expected = uint64(b.N)
	}
	if calls.Load() != expected {
		b.Fatalf("loader calls = %d, want %d", calls.Load(), expected)
	}
	b.ReportMetric(float64(calls.Load())/float64(b.N), "loads/batch")
	b.ReportMetric(benchmarkConcurrentRequests, "reads/batch")
	b.ReportMetric(benchmarkConcurrentWorkers, "goroutines/batch")
